# Gaussian Elimination

import numpy as np

n = int(input("Enter number of unknowns: "))

# Augmented matrix
A = np.zeros((n, n + 1))

print("Enter the augmented matrix:")
for i in range(n):
    for j in range(n + 1):
        A[i, j] = float(input(f"A[{i+1},{j+1}] = "))

print("\nAugmented matrix:")
print(A)

# Forward elimination
for k in range(n - 1):

    # If pivot is zero, interchange rows
    if A[k, k] == 0:
        for i in range(k + 1, n):
            if A[i, k] != 0:
                A[[k, i]] = A[[i, k]]
                break

    # Make elements below pivot zero
    for i in range(k + 1, n):
        m = A[i, k] / A[k, k]
        A[i] = A[i] - m * A[k]

print("\nUpper triangular matrix:")
print(A)

# Back substitution
x = np.zeros(n)

for i in range(n - 1, -1, -1):
    s = 0

    for j in range(i + 1, n):
        s += A[i, j] * x[j]

    x[i] = (A[i, n] - s) / A[i, i]

# Solution
print("\nSolution:")
for i in range(n):
    print(f"x{i+1} = {x[i]}")

# GAUSS SEIDEL METHOD

import numpy as np

n = int(input("Enter no. of unknowns = no. of equations: "))

A = np.zeros((n, n))

print("Enter coefficient matrix A:")
for i in range(n):
    for j in range(n):
        A[i, j] = float(input(f"Enter a{i+1}{j+1}: "))

print("Coefficient matrix A:")
print(A)


b = np.zeros(n)

print("Enter right side values:")
for i in range(n):
    b[i] = float(input(f"Enter b{i+1}: "))

print("Right side vector b:")
print(b)


# Check diagonal dominance
diagonal_dominant = True

for i in range(n):

    diag_element = abs(A[i, i])

    off_diag_sum = 0

    for j in range(n):
        if j != i:
            off_diag_sum += abs(A[i, j])

    if diag_element <= off_diag_sum:
        diagonal_dominant = False


if diagonal_dominant:
    print("Matrix A is diagonally dominant.")
    print("Gauss-Seidel method is expected to converge.")
else:
    print("Matrix A is not diagonally dominant.")
    print("Convergence is not guaranteed.")


# Initial approximation
x = np.zeros(n)

eps = float(input("Enter accuracy level: "))
N = int(input("Enter maximum iterations: "))


# Gauss-Seidel iteration
for k in range(N):

    x_old = x.copy()

    for i in range(n):

        p = b[i]

        for j in range(n):
            if j != i:
                p = p - A[i, j] * x[j]

        x[i] = p / A[i, i]

    error = np.max(np.abs(x - x_old))

    print(f"Iteration {k+1}: {x}")

    if error < eps:
        print("Gauss-Seidel method converged.")
        break

else:
    print("Maximum iterations reached.")


# Final solution
print("\nSolution:")

for i in range(n):
    print(f"x{i+1} = {x[i]}")

# GAUSS JORDAN FOR MATRIX INVERSE

import numpy as np

n = int(input("Enter order of matrix: "))

# Enter matrix
A = np.zeros((n, n))

print("Enter matrix A:")
for i in range(n):
    for j in range(n):
        A[i, j] = float(input(f"A[{i+1},{j+1}] = "))

# Augment A with identity matrix
aug = np.hstack((A, np.eye(n)))

# Gauss-Jordan elimination
for i in range(n):

    # Check zero pivot
    if aug[i, i] == 0:
        for k in range(i + 1, n):
            if aug[k, i] != 0:
                aug[[i, k]] = aug[[k, i]]
                break
        else:
            print("Inverse does not exist.")
            exit()

    # Make pivot = 1
    aug[i] = aug[i] / aug[i, i]

    # Make other elements in pivot column = 0
    for j in range(n):
        if i != j:
            m = aug[j, i]
            aug[j] = aug[j] - m * aug[i]

# Extract inverse
inverse = aug[:, n:]

print("\nInverse of A:")
print(inverse)

# House-Holder Transformation for upper triangular matrix

from math import *
import sys
k=1
t=1
e = 0.001
s_sq=0
n= int(input("Enter n, number of rows of matrix X:"))
p= int(input("Enter p, number of columns of matrix X:"))
x = [[0]*(p+1) for i in range(n+1)]
I = [[0]*(n+1) for i in range(n+1)]
H = [[0]*(n+1) for i in range(n+1)]
U = [[0]*(n+1) for i in range(n+1)]
u = [0 for i in range(n+1)]
sum1 = [0 for i in range(n+1)]
print("Input matrix X rowwise:")
for i in range(1,n+1):
    for j in range(1,p+1):
        xij=float(input("(i,j)th element="))
        x[i][j]=xij

for i in range(1,n+1):
    for j in range(1,n+1):
        if i == j:
            Iij = 1
        else:
            Iij = 0
        I[i][j] = Iij

while True:
    s_sq = 0 
    for i in range(t,n+1):
        s_sq += x[i][k]*x[i][k]
    if s_sq < e:
        k += 1
        continue
    if k > p:
        sys.exit()
    for i in range(1,n+1):
        if i < t:
            u[i]=0
        elif i == t:
            if x[t][k] < 0:
                u[t] = x[t][k] - sqrt(s_sq)
                beta = (-u[t])*sqrt(s_sq)
            else:
                u[t] = x[t][k] + sqrt(s_sq)
                beta = u[t]*sqrt(s_sq)
        else:
            u[i] = x[i][k]
    # computing H
    for i in range(1,n+1):
        for j in range(1,n+1):
            if i == j:
                U[i][j] = u[i]*u[i]/beta
            else:
                U[i][j] = u[i]*u[j]/beta
            H[i][j] = I[i][j]-U[i][j]
    # print H
    # computing HX
    for j in range(k+1,p+1):
        sum1[j]=0
        for i in range(t,n+1):
            sum1[j] += u[i]*x[i][j]

    for i in range(t,n+1):
        for j in range(k+1,p+1):
            x[i][j] = x[i][j] - (u[i]*sum1[j])/beta
    if x[t][k]<0:
        x[t][k] = sqrt(s_sq)
    else:
        x[t][k] = -sqrt(s_sq)
    for i in range(t+1,n+1):
        x[i][k]=0
    t=t+1
    k=k+1
    if k > p:
        break
print("The transformed matrix X is:")
for i in range(1,n+1):

        print(round(x[i][j],4),end='\t')

# GRAM SCHMIDT METHOD FOR ORTHOGONALISATION

from math import *

n = int(input("Enter number of rows: "))
p = int(input("Enter number of columns: "))

X = []

print("Enter matrix:")
for i in range(n):
    X.append(list(map(float, input().split())))

V = [[0] * p for i in range(n)]
T = [[0] * p for i in range(p)]

for k in range(p):

    # Find norm
    s = 0
    for i in range(n):
        s = s + X[i][k] ** 2

    d = sqrt(s)

    # Find unit vector
    for i in range(n):
        V[i][k] = X[i][k] / d

    # Find T
    for j in range(k, p):
        s = 0
        for i in range(n):
            s = s + V[i][k] * X[i][j]

        T[k][j] = s

    # Make remaining columns orthogonal
    for j in range(k + 1, p):
        for i in range(n):
            X[i][j] = X[i][j] - T[k][j] * V[i][k]

print("Matrix V:")
for i in range(n):
    for j in range(p):
        print(round(V[i][j], 4), end="\t")
    print()

print("Matrix T:")
for i in range(p):
    for j in range(p):
        print(round(T[i][j], 4), end="\t")
    print()
